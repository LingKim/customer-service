## 一、项目概述

### 1.1 功能范围

企业完成审核开通后，在后台接入第一个客服渠道：

1.  支持官网 / 网站、微信公众号、微信小程序三类渠道；

2.  渠道创建后自动生成渠道 ID 与渠道密钥；

3.  密钥支持脱敏展示、复制完整内容、重新生成；

4.  渠道列表展示渠道、状态、创建人、创建时间，并支持禁用/启用；

5.  已接入渠道的企业重新登录直接进入后台。

``` code-block-container
package cn.net.susan.user.controller;

import cn.net.susan.common.api.ApiResponse;
import cn.net.susan.user.entity.SysUser;
import cn.net.susan.user.mapper.SysUserMapper;
import com.baomidou.mybatisplus.core.conditions.update.LambdaUpdateWrapper;
import com.baomidou.mybatisplus.core.toolkit.Wrappers;
import jakarta.validation.constraints.NotBlank;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import java.time.LocalDateTime;
import java.util.Arrays;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

/**
 * 内部接口：仅供 tenant-service 审核通过后回填用户租户编码。
 */
@RestController
@RequestMapping("/api/user/internal")
public class UserInternalController {

    private final SysUserMapper sysUserMapper;

    public UserInternalController(SysUserMapper sysUserMapper) {
        this.sysUserMapper = sysUserMapper;
    }

    /**
     * 回填用户主归属租户编码。
     */
    @PutMapping("/users/{userId}/tenant-code")
    public ApiResponse<Void> updateTenantCode(
            @PathVariable long userId,
            @RequestBody BackfillBody body
    ) {
        LambdaUpdateWrapper<SysUser> update = Wrappers.lambdaUpdate(SysUser.class)
                .eq(SysUser::getId, userId)
                .eq(SysUser::getDeleted, false)
                .set(SysUser::getTenantCode, body.tenantCode())
                .set(SysUser::getUpdateTime, LocalDateTime.now());
        sysUserMapper.update(null, update);
        return ApiResponse.ok();
    }

    /**
     * 批量查询用户姓名（内部接口：跨服务展示创建人姓名）。
     */
    @GetMapping("/users/names")
    public ApiResponse<Map<String, String>> userNames(@RequestParam String ids) {
        List<Long> userIds = Arrays.stream(ids.split(","))
                .map(String::trim)
                .filter(s -> !s.isBlank())
                .map(Long::parseLong)
                .distinct()
                .toList();
        if (userIds.isEmpty()) {
            return ApiResponse.ok(Map.of());
        }
        Map<String, String> names = sysUserMapper.selectBatchIds(userIds).stream()
                .collect(Collectors.toMap(
                        u -> String.valueOf(u.getId()),
                        u -> u.getName() == null ? "" : u.getName(),
                        (a, b) -> a));
        return ApiResponse.ok(names);
    }

    /**
     * 请求体。
     */
    public record BackfillBody(
            @NotBlank(message = "租户编码不能为空")
            String tenantCode
    ) {
    }
}
```

``` code-block-container
package cn.net.susan.customer.internal;

import cn.net.susan.common.api.ApiResponse;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.ParameterizedTypeReference;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;

import java.util.List;
import java.util.Map;

/**
 * user-service 内部客户端：批量查询用户姓名，用于展示创建人姓名。
 */
@Component
public class UserNameClient {

    private final RestClient restClient;

    public UserNameClient(@Value("${yunti.user.internal-base-url}") String baseUrl) {
        this.restClient = RestClient.builder().baseUrl(baseUrl).build();
    }

    public Map<String, String> namesOf(List<String> userIds) {
        List<String> ids = userIds.stream().distinct().filter(s -> s != null && !s.isBlank()).toList();
        if (ids.isEmpty()) {
            return Map.of();
        }
        try {
            ApiResponse<Map<String, String>> response = restClient.get()
                    .uri("/api/user/internal/users/names?ids={ids}", String.join(",", ids))
                    .retrieve()
                    .body(new ParameterizedTypeReference<>() {
                    });
            return response == null || response.data() == null ? Map.of() : response.data();
        } catch (Exception e) {
            return Map.of();
        }
    }
}
```

``` code-block-container
package cn.net.susan.customer.service;

import cn.net.susan.common.auth.LoginUser;
import cn.net.susan.common.exception.BizException;
import cn.net.susan.common.id.SnowflakeIdGenerator;
import cn.net.susan.customer.entity.Channel;
import cn.net.susan.customer.entity.ChannelKey;
import cn.net.susan.customer.entity.ChannelTestLog;
import cn.net.susan.customer.entity.SkillGroup;
import cn.net.susan.customer.mapper.ChannelKeyMapper;
import cn.net.susan.customer.mapper.ChannelMapper;
import cn.net.susan.customer.mapper.ChannelTestLogMapper;
import cn.net.susan.customer.mapper.SkillGroupMapper;
import cn.net.susan.customer.internal.UserNameClient;
import com.baomidou.mybatisplus.core.toolkit.Wrappers;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.security.SecureRandom;
import java.time.LocalDateTime;
import java.util.List;
import java.util.Map;
import java.util.UUID;

/**
 * 企业渠道接入：创建渠道、生成密钥、联调测试、绑定技能组上线。
 */
@Service
public class ChannelOnboardingService {

    private static final SecureRandom RANDOM = new SecureRandom();

    private final ChannelMapper channelMapper;
    private final ChannelKeyMapper channelKeyMapper;
    private final SkillGroupMapper skillGroupMapper;
    private final ChannelTestLogMapper testLogMapper;
    private final SnowflakeIdGenerator idGenerator;
    private final UserNameClient userNameClient;

    public ChannelOnboardingService(
            ChannelMapper channelMapper,
            ChannelKeyMapper channelKeyMapper,
            SkillGroupMapper skillGroupMapper,
            ChannelTestLogMapper testLogMapper,
            SnowflakeIdGenerator idGenerator,
            UserNameClient userNameClient
    ) {
        this.channelMapper = channelMapper;
        this.channelKeyMapper = channelKeyMapper;
        this.skillGroupMapper = skillGroupMapper;
        this.testLogMapper = testLogMapper;
        this.idGenerator = idGenerator;
        this.userNameClient = userNameClient;
    }

    @Transactional
    public ChannelResult create(LoginUser user, int channelType, String name, String desc) {
        String tenant = tenantOf(user);
        if (channelType < 1 || channelType > 3) {
            throw new BizException(40001, "当前仅支持官网 / 微信公众号 / 微信小程序渠道");
        }
        if (name == null || name.isBlank() || name.length() > 64) {
            throw new BizException(40001, "请填写渠道名称");
        }

        SkillGroup skillGroup = ensureDefaultSkillGroup(user.userId(), tenant);
        long channelPk = idGenerator.nextId();
        String channelId = "CH" + UUID.randomUUID().toString().replace("-", "").substring(0, 30);
        Channel channel = Channel.builder()
                .id(channelPk)
                .tenantCode(tenant)
                .channelId(channelId)
                .channelType(channelType)
                .name(name.trim())
                .desc(desc == null || desc.isBlank() ? null : desc.trim())
                .skillGroupId(skillGroup.getId())
                .status(1)
                .stage(3)
                .isEnabled(true)
                // creator 存创建人用户 ID（口径 A），姓名由 user-service 关联展示
                .creator(String.valueOf(user.userId()))
                .deleted(false)
                .build();
        channelMapper.insert(channel);

        String appKey = randomKey();
        channelKeyMapper.insert(ChannelKey.builder()
                .id(idGenerator.nextId())
                .tenantCode(tenant)
                .channelId(channelPk)
                .appKey(appKey)
                .keyType(1)
                .status(1)
                .creator(String.valueOf(user.userId()))
                .deleted(false)
                .build());
        return toResult(channel, appKey, skillGroup, user.name());
    }

    @Transactional
    public List<ChannelResult> list(LoginUser user) {
        String tenant = tenantOf(user);
        SkillGroup group = defaultGroup(tenant);
        List<Channel> rows = channelMapper.selectList(
                        Wrappers.<Channel>lambdaQuery()
                                .eq(Channel::getTenantCode, tenant)
                                .eq(Channel::getDeleted, false)
                                .orderByDesc(Channel::getCreateTime));
        // 流程已简化，启用渠道统一视为运行中，兼容历史 stage=1/2 数据
        rows.stream()
                .filter(c -> c.getStatus() != null && c.getStatus() == 1 && c.getStage() != null && c.getStage() != 3)
                .forEach(c -> {
                    c.setStage(3);
                    channelMapper.updateById(c);
                });
        Map<String, String> names = userNameClient.namesOf(rows.stream().map(Channel::getCreator).toList());
        return rows.stream()
                .map(c -> toResult(c, appKeyOf(c.getId()), group,
                        names.getOrDefault(c.getCreator(), c.getCreator())))
                .toList();
    }

    @Transactional
    public ChannelResult rotateKey(LoginUser user, long channelId) {
        String tenant = tenantOf(user);
        Channel channel = requireOwned(channelId, tenant);
        List<ChannelKey> keys = channelKeyMapper.selectList(
                Wrappers.<ChannelKey>lambdaQuery()
                        .eq(ChannelKey::getChannelId, channel.getId())
                        .eq(ChannelKey::getStatus, 1)
                        .eq(ChannelKey::getDeleted, false));
        for (ChannelKey key : keys) {
            key.setStatus(2);
            key.setRotatedAt(LocalDateTime.now());
            key.setEditor(String.valueOf(user.userId()));
            key.setUpdateTime(LocalDateTime.now());
            channelKeyMapper.updateById(key);
        }
        String appKey = randomKey();
        channelKeyMapper.insert(ChannelKey.builder()
                .id(idGenerator.nextId())
                .tenantCode(tenant)
                .channelId(channel.getId())
                .appKey(appKey)
                .keyType(1)
                .status(1)
                .creator(String.valueOf(user.userId()))
                .deleted(false)
                .build());
        return toResult(channel, appKey, defaultGroup(tenant), user.name());
    }

    @Transactional
    public ChannelResult updateStatus(LoginUser user, long channelId, boolean enabled) {
        String tenant = tenantOf(user);
        Channel channel = requireOwned(channelId, tenant);
        channel.setStatus(enabled ? 1 : 2);
        channel.setIsEnabled(enabled);
        channel.setEditor(String.valueOf(user.userId()));
        channel.setUpdateTime(LocalDateTime.now());
        channelMapper.updateById(channel);
        return toResult(channel, appKeyOf(channelId), defaultGroup(tenant), user.name());
    }

    private SkillGroup ensureDefaultSkillGroup(long userId, String tenant) {
        SkillGroup group = defaultGroup(tenant);
        if (group != null) {
            return group;
        }
        SkillGroup created = SkillGroup.builder()
                .id(idGenerator.nextId())
                .tenantCode(tenant)
                .name("默认技能组")

                Wrappers.<SkillGroup>lambdaQuery()
                        .eq(SkillGroup::getTenantCode, tenant)

                        .orderByAsc(SkillGroup::getId)
                        .last("LIMIT 1"));

        if (channel == null || !tenant.equals(channel.getTenantCode()) || Boolean.TRUE.equals(channel.getDeleted())) {
            throw new BizException(40401, "渠道不存在");
        }
        return channel;

                Wrappers.<ChannelKey>lambdaQuery()
                        .eq(ChannelKey::getChannelId, channelPk)
                        .eq(ChannelKey::getStatus, 1)
                        .eq(ChannelKey::getDeleted, false)
                        .last("LIMIT 1"));
```

## 十、总结

本文档完成了企业审核开通后的第一个业务闭环：渠道接入与密钥生成。

### 完成的重点

1.  **渠道模型**：channel 记录渠道主体与状态，channel\_key 独立保存密钥，skill\_group 为默认接待技能组；

2.  **租户隔离**：所有渠道数据按 tenant\_code 隔离，正式租户（T 开头）才允许接入；

3.  **密钥安全**：页面只展示脱敏密钥，复制时获取完整内容；支持重新生成并吊销旧密钥；

4.  **公共字段口径**：creator 存创建人用户 ID，姓名通过 user-service 内部接口关联返回，保持全系统可审计；

5.  **页面闭环**：渠道列表 → 创建渠道 → 获取密钥 → 完成接入；

6.  **登录体验**：已存在生效渠道的企业重新登录直接进入后台，不再重复进入开通引导。
