## 一、项目概述

前面已经实现了企业用户注册、完整资料、审核和渠道接入、密钥生成。

今天接着实现企业新手引导中的第二步：智能机器人配置功能。

### 1.1 功能范围

实现企业机器人的核心可落库配置：

1.  机器人设置：名称、欢迎语、兜底话术、转人工提示、服务开关；

2.  意图管理：默认意图、新建/编辑、启用/停用、训练语料；

3.  模型选择：千问 Plus / DeepSeek Chat / 千问 VL，temperature 参数保存。

### 1.2 技术选型

|                    |                 |
|--------------------|-----------------|
| 技术               | 说明            |
| Python 3.13        | 后端脚本运行    |
| FastAPI            | HTTP 服务       |
| psycopg 3          | PostgreSQL 驱动 |
| PostgreSQL ai\_db  | 机器人数据      |
| Vue 3 + TypeScript | 前端            |

### 1.3 接口清单

|          |                             |                              |
|----------|-----------------------------|------------------------------|
| 方法     | 路径                        | 说明                         |
| GET/POST | /api/ai/v1/bot/intents      | 意图列表 / 新建意图          |
| PUT      | /api/ai/v1/bot/intents/{id} | 编辑意图 / 启停              |
| GET/PUT  | /api/ai/v1/bot/settings     | 机器人设置                   |
| GET      | /api/ai/v1/bot/models       | 模型列表（自动初始化默认值） |
| PUT      | /api/ai/v1/bot/model        | 选择模型并保存温度           |

### 1.4 与上一篇的关系

|                                               |                                |
|-----------------------------------------------|--------------------------------|
| 已有模块                                      | 本次处理                       |
| yunti-ai 骨架                                 | 新增数据库连接与机器人路由     |

"""机器人配置接口：意图管理、机器人设置、模型选择。"""

from __future__ import annotations

from datetime import datetime
from decimal import Decimal
import random
import time
from typing import Any, Optional

from fastapi import APIRouter, Header, HTTPException
from pydantic import BaseModel, ConfigDict, Field
from pydantic.alias_generators import to_camel

from ..core.db import connect

router = APIRouter(prefix="/ai/v1/bot", tags=["bot"])


class ApiModel(BaseModel):
    model_config = ConfigDict(alias_generator=to_camel, populate_by_name=True)

DEFAULT_INTENTS = [
    ("查询物流", "我的快递到哪了\n帮我查一下订单物流\n订单什么时候发货"),
    ("退款售后", "我要退款\n怎么申请售后\n商品有问题怎么退"),
    ("价格优惠", "现在有什么优惠\n能便宜一点吗\n有没有优惠券"),
    ("人工客服", "转人工\n找人工客服\n我要投诉"),
    ("开票服务", "怎么开发票\n发票抬头怎么填\n需要开增值税发票"),
]


def ok(data: Any = None):
    return {"code": 0, "message": "成功", "data": data}


def clean(row: Optional[dict]) -> Optional[dict]:
    if row is None:
        return None
    result = {}
    for key, value in row.items():
        key = {
            "bot_name": "botName",
            "welcome_message": "welcomeMessage",
            "fallback_message": "fallbackMessage",
            "transfer_prompt": "transferPrompt",
            "is_enabled": "isEnabled",
            "model_key": "modelKey",
            "model_name": "modelName",
            "model_type": "modelType",
            "intent_code": "intentCode",
            "hit_count": "hitCount",
            "tenant_code": "tenantCode",
            "update_time": "updateTime",
        }.get(key, key)
        if isinstance(value, Decimal):
            result[key] = float(value)
        elif isinstance(value, datetime):
            result[key] = value.strftime("%Y-%m-%d %H:%M:%S")
        else:
            result[key] = value
    return result


def require_tenant(tenant_code: Optional[str]) -> str:
    if not tenant_code or tenant_code == "PLATFORM":
        raise HTTPException(status_code=400, detail="缺少企业租户上下文")
    return tenant_code


def new_id() -> int:
    return int(time.time() * 1000) * 1024 + random.randint(0, 1023)


class IntentCreate(ApiModel):
    name: str = Field(min_length=1, max_length=64)
    samples: Optional[str] = None


class IntentUpdate(ApiModel):
    name: Optional[str] = Field(default=None, max_length=64)
    samples: Optional[str] = None
    status: Optional[int] = Field(default=None, ge=1, le=2)


class SettingUpdate(ApiModel):
    bot_name: str = Field(min_length=1, max_length=64)
    welcome_message: Optional[str] = Field(default=None, max_length=255)
    fallback_message: Optional[str] = Field(default=None, max_length=255)
    transfer_prompt: Optional[str] = Field(default=None, max_length=255)
    is_enabled: bool = True


class ModelUpdate(ApiModel):
    model_key: str = Field(min_length=1, max_length=64)
    model_name: Optional[str] = Field(default=None, max_length=64)
    provider: Optional[str] = Field(default=None, max_length=32)
    model_type: Optional[int] = Field(default=1, ge=1, le=3)
    temperature: float = Field(default=0.3, ge=0, le=2)


@router.get("/intents")
def list_intents(x_tenant_code: Optional[str] = Header(default=None)):
    tenant = require_tenant(x_tenant_code)
    with connect() as conn:
        count = conn.execute("SELECT COUNT(1) FROM bot_intent WHERE tenant_code=%s AND is_deleted=FALSE", (tenant,)).fetchone()
        if count and count["count"] == 0:
            conn.cursor().executemany(
                """INSERT INTO bot_intent
                      (id, tenant_code, intent_code, name, status, confidence, hit_count, samples)
                   VALUES (%s, %s, %s, %s, 1, 0.00, 0, %s)""",
                [(new_id(), tenant, f"IT000{i + 1}", name, samples) for i, (name, samples) in enumerate(DEFAULT_INTENTS)],
            )
            conn.commit()
        rows = conn.execute(
            """SELECT id, tenant_code, intent_code, name, status, confidence, hit_count, samples, update_time
                 FROM bot_intent
                WHERE tenant_code=%s AND is_deleted=FALSE
                ORDER BY create_time DESC""",
            (tenant,),
        ).fetchall()
    return ok([clean(r) for r in rows])


@router.post("/intents")
def create_intent(body: IntentCreate, x_tenant_code: Optional[str] = Header(default=None)):
    tenant = require_tenant(x_tenant_code)
    intent_code = "IT" + datetime.now().strftime("%Y%m%d%H%M%S")
    with connect() as conn, conn.cursor() as cur:
        row = cur.execute(
            """INSERT INTO bot_intent (id, tenant_code, intent_code, name, status, confidence, hit_count, samples)
               VALUES (%s, %s, %s, %s, 1, NULL, 0, %s)
               RETURNING id, tenant_code, intent_code, name, status, confidence, hit_count, samples, update_time""",
            (new_id(), tenant, intent_code, body.name.strip(), body.samples),
        ).fetchone()
        conn.commit()
    return ok(clean(row))


@router.put("/intents/{intent_id}")
def update_intent(intent_id: int, body: IntentUpdate, x_tenant_code: Optional[str] = Header(default=None)):
    tenant = require_tenant(x_tenant_code)
    sets = []
    params = []
    if body.name is not None:
        sets.append("name=%s")
        params.append(body.name.strip())
    if body.samples is not None:
        sets.append("samples=%s")
        params.append(body.samples)
    if body.status is not None:
        sets.append("status=%s")
        params.append(body.status)
    if not sets:
        return ok()
    sets.append("update_time=CURRENT_TIMESTAMP")
    params += [intent_id, tenant]
    with connect() as conn, conn.cursor() as cur:
        cur.execute(f"UPDATE bot_intent SET {', '.join(sets)} WHERE id=%s AND tenant_code=%s AND is_deleted=FALSE", params)
        conn.commit()
    return ok()


@router.get("/settings")
def get_setting(x_tenant_code: Optional[str] = Header(default=None)):
    tenant = require_tenant(x_tenant_code)
    with connect() as conn:
        row = conn.execute(
            """SELECT id, tenant_code, bot_name, welcome_message, fallback_message, transfer_prompt,
                      is_enabled, model_key, temperature, update_time
                 FROM bot_setting
                WHERE tenant_code=%s AND is_deleted=FALSE""",
            (tenant,),
        ).fetchone()
        if row is None:
            row = conn.execute(
                """INSERT INTO bot_setting
                      (id, tenant_code, bot_name, welcome_message, fallback_message,
                       transfer_prompt, is_enabled, model_key, temperature)
                   VALUES (%s, %s, '小云', %s, %s, %s, TRUE, 'qwen-plus', 0.35)
                   RETURNING id, tenant_code, bot_name, welcome_message, fallback_message,
                             transfer_prompt, is_enabled, model_key, temperature, update_time""",
                (
                    new_id(),
                    tenant,
                    "您好，我是云梯智能客服小云，很高兴为您服务。您可以直接描述问题，例如“查询物流”“退款怎么办理”。",
                    "抱歉，我暂时无法准确回答您的问题，已为您记录并建议转人工客服进一步处理。",
                    "如需人工客服，请回复“转人工”，或通过右上角电话联系我们。",
                ),
            ).fetchone()
            conn.commit()
    return ok(clean(row))


@router.put("/settings")
def update_setting(body: SettingUpdate, x_tenant_code: Optional[str] = Header(default=None)):
    tenant = require_tenant(x_tenant_code)
    get_setting(x_tenant_code)
    with connect() as conn, conn.cursor() as cur:
        cur.execute(
            """UPDATE bot_setting
                  SET bot_name=%s, welcome_message=%s, fallback_message=%s, transfer_prompt=%s,
                      is_enabled=%s, update_time=CURRENT_TIMESTAMP
                WHERE tenant_code=%s AND is_deleted=FALSE""",
            (body.bot_name.strip(), body.welcome_message, body.fallback_message, body.transfer_prompt, body.is_enabled, tenant),
        )
        conn.commit()
    return ok(get_setting(x_tenant_code)["data"])


@router.get("/models")
def list_models(x_tenant_code: Optional[str] = Header(default=None)):
    tenant = require_tenant(x_tenant_code)
    defaults = [
        ("qwen-plus", "千问 Plus", "qwen", 1, 0.35),
        ("deepseek-chat", "DeepSeek Chat", "deepseek", 1, 0.30),
        ("qwen-vl-plus", "千问 VL", "qwen", 2, 0.20),
    ]
    with connect() as conn, conn.cursor() as cur:
        cur.executemany(
            """INSERT INTO bot_model (id, tenant_code, model_key, model_name, provider, model_type, temperature, is_enabled)
               VALUES (%s, %s, %s, %s, %s, %s, %s, TRUE)
               ON CONFLICT (tenant_code, model_key) DO NOTHING""",
            [(new_id(), tenant, k, n, p, t, temp) for k, n, p, t, temp in defaults],
        )
        conn.commit()
        rows = conn.execute(
            """SELECT id, tenant_code, model_key, model_name, provider, model_type, temperature, is_enabled
                 FROM bot_model
                WHERE tenant_code=%s AND is_deleted=FALSE
                ORDER BY model_type, id""",
            (tenant,),
        ).fetchall()
    return ok([clean(r) for r in rows])


@router.put("/model")
def update_model(body: ModelUpdate, x_tenant_code: Optional[str] = Header(default=None)):
    tenant = require_tenant(x_tenant_code)
    provider = body.provider or "yunti"
    with connect() as conn, conn.cursor() as cur:
        cur.execute(
            """INSERT INTO bot_model (id, tenant_code, model_key, model_name, provider, model_type, temperature, is_enabled)
               VALUES (%s, %s, %s, %s, %s, %s, %s, TRUE)
               ON CONFLICT (tenant_code, model_key)
               DO UPDATE SET model_name=EXCLUDED.model_name, provider=EXCLUDED.provider,
                             model_type=EXCLUDED.model_type, temperature=%s,
                             is_enabled=TRUE, update_time=CURRENT_TIMESTAMP
               RETURNING id, tenant_code, model_key, model_name, provider, model_type, temperature, is_enabled""",
            (new_id(), tenant, body.model_key, body.model_name or body.model_key, provider, body.model_type, body.temperature, body.temperature),
        ).fetchone()
        cur.execute(
            """UPDATE bot_setting SET model_key=%s, temperature=%s, update_time=CURRENT_TIMESTAMP
              WHERE tenant_code=%s AND is_deleted=FALSE""",
            (body.model_key, body.temperature, tenant),
        )
        conn.commit()
    return ok()

#!/usr/bin/env bash
# ============================================================================
# yunti-ai · AI 编排中心启动脚本（参考企业知识库后端 start.sh 方案）
#
# 用法:
#   ./start.sh                     # 默认 127.0.0.1:9100
#   ./start.sh --host 0.0.0.0      # 监听所有网卡

stop_port() {
  local port="$1"
  local pids
  pids=$(lsof -ti tcp:"$port" 2>/dev/null || true)
  if [ -n "$pids" ]; then
    echo -e "  ${YELLOW}发现端口 $port 占用进程: $pids，正在停止...${NC}"
    for attempt in 1 2 3 4 5; do
      # shellcheck disable=SC2086
      kill $pids 2>/dev/null || true
      sleep 1
      pids=$(lsof -ti tcp:"$port" 2>/dev/null || true)
      [ -z "$pids" ] && break
    done
  fi
  pids=$(lsof -ti tcp:"$port" 2>/dev/null || true)
  if [ -n "$pids" ]; then
    echo -e "  ${YELLOW}进程仍未退出，执行强制停止...${NC}"
    # shellcheck disable=SC2086
    kill -9 $pids 2>/dev/null || true
    sleep 1
  fi
  if command -v pkill >/dev/null 2>&1; then
    pkill -9 -f "$PROJECT_DIR/.venv/bin/python" 2>/dev/null || true
  fi
  sleep 1
}

import { request } from '../request'

export interface BotIntent {
  id: number
  intentCode: string
  name: string
  status: number
  confidence?: number | null
  hitCount: number
  samples?: string
  updateTime?: string
}

export interface BotSetting {
  botName: string
  welcomeMessage?: string
  fallbackMessage?: string
  transferPrompt?: string
  isEnabled: boolean
  modelKey?: string
  temperature: number
}

export interface BotModel {
  id: number
  modelKey: string
  modelName: string
  provider: string
  modelType: number
  temperature: number
  isEnabled: boolean
}

export function listBotIntents(): Promise<BotIntent[]> {
  return request({ url: '/ai/v1/bot/intents', method: 'get' })
}

export function createBotIntent(data: { name: string; samples?: string }): Promise<BotIntent> {
  return request({ url: '/ai/v1/bot/intents', method: 'post', data })
}

export function updateBotIntent(
  id: number,
  data: { name?: string; samples?: string; status?: number },
): Promise<void> {
  return request({ url: `/ai/v1/bot/intents/${id}`, method: 'put', data })
}

export function getBotSetting(): Promise<BotSetting> {
  return request({ url: '/ai/v1/bot/settings', method: 'get' })
}

export function saveBotSetting(data: Partial<BotSetting>): Promise<BotSetting> {
  return request({ url: '/ai/v1/bot/settings', method: 'put', data })
}

export function listBotModels(): Promise<BotModel[]> {
  return request({ url: '/ai/v1/bot/models', method: 'get' })
}

export function selectBotModel(data: { modelKey: string; temperature: number }): Promise<void> {
  return request({ url: '/ai/v1/bot/model', method: 'put', data })

import { onMounted, reactive, ref } from 'vue'

const saving = ref(false)
const savingIntent = ref(false)
const savingModel = ref(false)

const intentForm = reactive({ name: '', samples: '' })

onMounted(loadAll)

async function loadAll() {
  try {
    const settingData = await getBotSetting()
    Object.assign(setting, settingData)
    temperature.value = Number(settingData.temperature || 0.3)
    selectedModelKey.value = settingData.modelKey || ''
  } catch { /* ignore */ }
  await Promise.all([loadIntents(), loadModels()])
}

async function loadIntents() {
  loadingIntents.value = true
  try {
    intents.value = await listBotIntents()
  } finally {
    loadingIntents.value = false
  }

  models.value = await listBotModels()
}

function openIntent(row?: BotIntent) {
  editingIntent.value = row || null
  intentForm.name = row?.name || ''
  intentForm.samples = row?.samples || ''
  intentVisible.value = true
}

async function saveIntent() {
  if (!intentForm.name.trim()) {
    ElMessage.warning('请填写意图名称')
    return
  }
  savingIntent.value = true
  try {
    if (editingIntent.value) {
      await updateBotIntent(editingIntent.value.id, { name: intentForm.name, samples: intentForm.samples })
    } else {
      await createBotIntent({ name: intentForm.name, samples: intentForm.samples })
    }
    intentVisible.value = false
    ElMessage.success('保存成功')
    await loadIntents()
  } finally {
    savingIntent.value = false
  }
}

async function toggleIntent(row: BotIntent) {
  await updateBotIntent(row.id, { status: row.status === 1 ? 2 : 1 })
  row.status = row.status === 1 ? 2 : 1
  ElMessage.success(row.status === 1 ? '意图已启用' : '意图已停用')
}

async function saveSettings() {
  saving.value = true
  try {
    const data = await saveBotSetting({ ...setting })
    Object.assign(setting, data)
    ElMessage.success('设置已保存')
  } finally {
    saving.value = false
  }
}

async function saveModel() {
  if (!selectedModelKey.value) {
    ElMessage.warning('请选择模型')
    return
  }
  savingModel.value = true
  try {
    await selectBotModel({ modelKey: selectedModelKey.value, temperature: temperature.value })
    setting.modelKey = selectedModelKey.value
    setting.temperature = temperature.value
    ElMessage.success('模型配置已保存')
  } finally {
    savingModel.value = false
  }
}

function typeText(type: number) {

  if (type === 2) return '视觉'
  return '轻量'
}

## 十、总结

本文实现了机器人配置的最小完整闭环：意图、话术、模型均按 tenant\_code 隔离落库。

后续可在该基础上接入 LangGraph Agent、LlamaIndex 知识检索与真实大模型供应商。

### 完成的重点

1.  **机器人领域建模**：意图、模型、设置拆成三张独立表，配置之间职责清晰；模型与意图带唯一键约束，设置每租户唯一，避免出现重复配置。

2.  **租户级数据隔离**：接口统一从 `X-Tenant-Code` 请求头读取企业上下文，所有读写都带 `tenant_code`，`PLATFORM` 等平台上下文被拒绝，从数据入口杜绝跨企业串数据。

3.  **默认数据懒初始化**：企业第一次打开页面时自动写入 5 条默认意图、3 个可用模型和默认欢迎语，企业侧零配置也能看到完整页面效果，不需要手工造数。

4.  **机器人设置闭环**：机器人名称、欢迎语、未识别兜底话术、转人工提示、服务开关，保存后立即落库并可回读。

5.  **意图管理闭环**：默认意图、新建意图、编辑训练语料、启用/停用全部可用，列表带序号、状态、置信度与命中次数。

6.  **模型选择闭环**：支持千问 Plus、DeepSeek Chat、千问 VL，选择模型并保存 temperature 后同步写入机器人当前设置，页面刷新不回退。

7.  **全链路打通**：网关新增 `/api/ai` 路由、前端 Vite 代理、真实页面替换占位页；同时升级启动脚本支持 `restart` 与更强端口清理，解决 uvicorn reload 残留进程导致的 `Address already in use`。
