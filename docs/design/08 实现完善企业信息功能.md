## 一、项目概述

### 1.1 功能范围

本文档实现企业资料的完善、营业执照上传与审核状态查看，是企业账号完成注册后的下一个环节：

1. 企业用户填写营业执照号、注册地址、法人等资料；
2. 营业执照上传到对象存储（local / RustFS）并在 file\_meta 落库；
3. 提交企业资料，写入 enterprise 与 enterprise\_review；
4. 前端提供企业开通引导页与企业信息查看页。

### 1.2 技术选型

| 技术 | 版本 | 说明 |
| --- | --- | --- |
| JDK | 21+ | 验证环境 JDK 25 |
| Spring Boot | 4.0.3 | 基础框架 |
| MyBatis-Plus | 3.5.17 | ORM |

| Spring Cloud Gateway | 5.0.2 | 网关 |
| Vue 3 + TypeScript | \- | 前端 |

```java
package cn.net.susan.tenant.security;

import cn.net.susan.common.api.ResultCode;
import cn.net.susan.common.auth.LoginUser;
import cn.net.susan.common.exception.BizException;
import io.jsonwebtoken.Claims;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Component;

import javax.crypto.SecretKey;
import java.nio.charset.StandardCharsets;

/**
 * JWT 解析（HS256，与 user-service 共用同一签名密钥）。
 */
@Component
public class JwtTokenParser {

    private final SecretKey secretKey;

    public JwtTokenParser(@Value("${yunti.auth.jwt-secret}") String secret) {
        this.secretKey = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
    }

    /**
     * 解析令牌；非法 / 过期抛未认证异常。
     */
    public LoginUser parseToken(String token) {
        try {
            Claims claims = Jwts.parser()
                    .verifyWith(secretKey)
                    .build()
                    .parseSignedClaims(token)
                    .getPayload();
            return new LoginUser(
                    Long.parseLong(claims.getSubject()),
                    claims.get("uno", String.class),
                    claims.get("name", String.class),
                    claims.get("type", Integer.class),
                    claims.get("tnt", String.class)
            );
        } catch (Exception e) {
            throw new BizException(ResultCode.UNAUTHORIZED);
        }
    }

    /**
     * 从 Authorization 请求头解析当前登录用户。
     */
    public LoginUser requireLoginUser(String authorization) {
        if (authorization == null || !authorization.startsWith("Bearer ")) {
            throw new BizException(ResultCode.UNAUTHORIZED);
        }
        return parseToken(authorization.substring(7));
    }
}
```

```java
        this.enterpriseReviewMapper = enterpriseReviewMapper;
        this.idGenerator = idGenerator;
    }

    /**
     * 获取当前企业开通引导状态。
     */
    @Transactional(readOnly = true)
    public GuideState getState(long applicantId) {
        Enterprise enterprise = requireEnterprise(applicantId);
        Optional<EnterpriseReview> reviewOpt = Optional.ofNullable(
                enterpriseReviewMapper.findLatestByEnterpriseId(enterprise.getId()));

        if (reviewOpt.isEmpty()) {
            return toState(enterprise, null, STAGE_PENDING_PROFILE);
        }
        EnterpriseReview review = reviewOpt.get();
        String stage = switch (review.getStatus()) {
            case 1 -> STAGE_PENDING_REVIEW;
            case 3 -> STAGE_REJECTED;
            default -> STAGE_APPROVED;
        };
        if (enterprise.getStatus() != null && enterprise.getStatus() == 2) {
            stage = STAGE_APPROVED;
        }
        return toState(enterprise, review, stage);
    }

    /**
     * 保存企业资料（不提交审核，供中途暂存）。
     */
    @Transactional
    public GuideState saveProfile(long applicantId, ProfileRequest request) {
        Enterprise enterprise = requireEnterprise(applicantId);
        updateEnterpriseProfile(enterprise.getId(), request);
        return getState(applicantId);
    }

    /**
     * 提交企业资料进入审核（可驳回后重提，版本号递增）。
     */
    @Transactional
    public GuideState submitReview(long applicantId, ProfileRequest request) {
        Enterprise enterprise = requireEnterprise(applicantId);
        if (enterprise.getStatus() != null && enterprise.getStatus() == 2) {
            throw new BizException(40001, "企业已开通，无需重复提交审核");
        }

        long reviewId = idGenerator.nextId();
        int versionNo = enterpriseReviewMapper.nextVersionNo(enterprise.getId());
        String applyNo = "YR" + APPLY_NO_TIME.format(LocalDateTime.now())
                + ThreadLocalRandom.current().nextInt(1000, 10000);

        EnterpriseReview review = EnterpriseReview.builder()
                .id(reviewId)
                .enterpriseId(enterprise.getId())
                .applyNo(applyNo)
                .versionNo(versionNo)
                .applicantId(applicantId)
                .companyName(request.companyName())
                .industry(request.industry())
                .scale(request.scale())
                .contactName(request.contactName())
                .contactPhone(request.contactPhone())
                .contactEmail(request.contactEmail())
                .licenseNo(request.licenseNo())
                .registerAddress(request.registerAddress())
                .legalPerson(request.legalPerson())
                .licenseFileId(request.licenseFileId())
                .status(1)
                .submitTime(LocalDateTime.now())
                .creator(String.valueOf(applicantId))
                .deleted(false)
                .build();
        enterpriseReviewMapper.insert(review);

        // 企业主数据同步为最新提交资料
        updateEnterpriseProfile(enterprise.getId(), request);
        return getState(applicantId);
    }

    private Enterprise requireEnterprise(long applicantId) {
        Enterprise enterprise = enterpriseMapper.findByApplicantId(String.valueOf(applicantId));
        if (enterprise == null) {
            throw new BizException(40401, "账号未关联企业资料，请通过注册流程开通企业");
        }
        return enterprise;
    }

    private void updateEnterpriseProfile(long enterpriseId, ProfileRequest request) {
        LambdaUpdateWrapper<Enterprise> update = Wrappers.lambdaUpdate(Enterprise.class)
                .eq(Enterprise::getId, enterpriseId)
                .eq(Enterprise::getDeleted, false)
                .set(Enterprise::getCompanyName, request.companyName())
                .set(Enterprise::getIndustry, request.industry())
                .set(Enterprise::getScale, request.scale())
                .set(Enterprise::getLicenseNo, request.licenseNo())
                .set(Enterprise::getRegisterAddress, request.registerAddress())
                .set(Enterprise::getLegalPerson, request.legalPerson())
                .set(Enterprise::getLicenseFileId, request.licenseFileId())
                .set(Enterprise::getContactName, request.contactName())
                .set(Enterprise::getContactPhone, request.contactPhone())
                .set(Enterprise::getContactEmail, request.contactEmail())
                .set(Enterprise::getEditor, "SELF_SERVICE")
                .set(Enterprise::getUpdateTime, LocalDateTime.now());
        enterpriseMapper.update(null, update);
    }

    private GuideState toState(
            Enterprise enterprise,
            EnterpriseReview review,
            String stage
    ) {
        return new GuideState(
                enterprise.getId(),
                enterprise.getEnterpriseCode(),
                stage,
                enterprise.getCompanyName(),
                enterprise.getIndustry(),
                enterprise.getScale(),
                enterprise.getLicenseNo(),
                enterprise.getRegisterAddress(),
                enterprise.getLegalPerson(),
                enterprise.getLicenseFileId() == null ? null : String.valueOf(enterprise.getLicenseFileId()),
                enterprise.getContactName(),
                enterprise.getContactPhone(),
                enterprise.getContactEmail(),
                review == null ? null : review.getApplyNo(),
                review == null ? null : review.getVersionNo(),
                review == null ? null : review.getRejectReason(),
                review == null || review.getSubmitTime() == null
                        ? null
                        : review.getSubmitTime().toString(),
                enterprise.getTenantCode()
        );
    }

    /**

    public record GuideState(
```

```java
 * 本地磁盘实现：仅用于开发联调，生产切换 file.storage.type=rustfs。
 */
@Component
@ConditionalOnProperty(name = "yunti.file.storage.type", havingValue = "local", matchIfMissing = true)
public class LocalObjectStorage implements ObjectStorage {

    private final Path root;

    public LocalObjectStorage(FileStorageProperties properties) {
        this.root = Paths.get(properties.getLocalDir()).toAbsolutePath().normalize();
    }

    @PostConstruct
    public void init() throws IOException {
        Files.createDirectories(root);
    }

    @Override
    public void upload(String key, InputStream input, long size, String mimeType) {
        try {
            Path target = resolve(key);
            Files.createDirectories(target.getParent());
            Files.copy(input, target, StandardCopyOption.REPLACE_EXISTING);
        } catch (IOException e) {
            throw new IllegalStateException("本地文件保存失败: " + key, e);
        }
    }

    @Override
    public InputStream download(String key) {
        try {
            return Files.newInputStream(resolve(key));
        } catch (IOException e) {
            throw new IllegalStateException("本地文件读取失败: " + key, e);
        }
    }

    @Override
    public void delete(String key) {
        try {
            Files.deleteIfExists(resolve(key));
        } catch (IOException e) {
            throw new IllegalStateException("本地文件删除失败: " + key, e);
        }
    }

    @Override
    public boolean exists(String key) {
        return Files.exists(resolve(key));
    }

    }
```

```java
import software.amazon.awssdk.services.s3.model.S3Exception;

import java.net.URI;
import java.time.Duration;

/**
 * RustFS / MinIO / S3 兼容对象存储实现（参考企业智能知识库 RustFsS3Config + RustFileStorage）。
 */
@Component
@ConditionalOnProperty(name = "yunti.file.storage.type", havingValue = "rustfs")
public class RustFsObjectStorage implements ObjectStorage {

    private static final Logger log = LoggerFactory.getLogger(RustFsObjectStorage.class);

    private final S3Client s3Client;
    private final String bucket;

    public RustFsObjectStorage(FileStorageProperties properties) {
        FileStorageProperties.Rustfs cfg = properties.getRustfs();
        String protocol = cfg.isSecure() ? "https" : "http";
        URI endpoint = URI.create(protocol + "://" + cfg.getEndpoint() + ":" + cfg.getPort());
        this.bucket = cfg.getBucket();
        this.s3Client = S3Client.builder()
                .endpointOverride(endpoint)
                .credentialsProvider(StaticCredentialsProvider.create(
                        AwsBasicCredentials.create(cfg.getAccessKey(), cfg.getSecretKey())))
                .region(Region.of("us-east-1"))
                .serviceConfiguration(S3Configuration.builder()
                        .checksumValidationEnabled(false)
                        .build())
                .overrideConfiguration(builder -> builder
                        .apiCallTimeout(Duration.ofSeconds(30))
                        .apiCallAttemptTimeout(Duration.ofSeconds(30)))
                .build();
    }

    @PostConstruct
    public void init() {
        try {
            s3Client.createBucket(builder -> builder.bucket(bucket));
        } catch (S3Exception e) {
            // Bucket 已存在时忽略
            log.debug("RustFS bucket 已存在或创建被忽略: {}", e.getMessage());
        }
    }

    @Override
    public void upload(String key, InputStream input, long size, String mimeType) {
        s3Client.putObject(
                PutObjectRequest.builder()
                        .bucket(bucket)
                        .key(key)
                        .contentType(mimeType)
                        .build(),
                RequestBody.fromInputStream(input, size));
        log.info("RustFS 上传成功: bucket={}, key={}", bucket, key);
    }

    @Override
    public InputStream download(String key) {
        ResponseInputStream<software.amazon.awssdk.services.s3.model.GetObjectResponse> response =
                s3Client.getObject(GetObjectRequest.builder().bucket(bucket).key(key).build());
        return response;
    }

    @Override
    public void delete(String key) {
        s3Client.deleteObject(DeleteObjectRequest.builder().bucket(bucket).key(key).build());
    }

    @Override
    public boolean exists(String key) {
        try {
            s3Client.headObject(HeadObjectRequest.builder().bucket(bucket).key(key).build());
            return true;
        } catch (S3Exception e) {
            return false;
        }
    }
}
```

```java
@Component
public class JwtTokenParser {

    private final SecretKey secretKey;

    public JwtTokenParser(@Value("${yunti.auth.jwt-secret}") String secret) {
        this.secretKey = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
    }

    public LoginUser requireLoginUser(String authorization) {
        if (authorization == null || !authorization.startsWith("Bearer ")) {
            throw new BizException(ResultCode.UNAUTHORIZED);
        }
        try {
            Claims claims = Jwts.parser()
                    .verifyWith(secretKey)
                    .build()
                    .parseSignedClaims(authorization.substring(7))
                    .getPayload();
            return new LoginUser(
                    Long.parseLong(claims.getSubject()),
                    claims.get("uno", String.class),
                    claims.get("name", String.class),
                    claims.get("type", Integer.class),
                    claims.get("tnt", String.class)
            );
        } catch (Exception e) {
            throw new BizException(ResultCode.UNAUTHORIZED);
        }
    }
```

```java
import org.springframework.stereotype.Service;

import org.springframework.web.multipart.MultipartFile;

import java.io.InputStream;
import java.time.LocalDateTime;

/**
 * 企业营业执照附件：上传到对象存储并记录 file_meta，供企业资料审核使用。
 */
@Service
public class EnterpriseLicenseFileService {

    private static final String LICENSE_PREFIX = "enterprise/license";

    private final ObjectStorage objectStorage;
    private final FileMetaMapper fileMetaMapper;
    private final FileStorageProperties properties;
    private final SnowflakeIdGenerator idGenerator;

    public EnterpriseLicenseFileService(
            ObjectStorage objectStorage,
            FileMetaMapper fileMetaMapper,
            FileStorageProperties properties,
            SnowflakeIdGenerator idGenerator
    ) {
        this.objectStorage = objectStorage;
        this.fileMetaMapper = fileMetaMapper;
        this.properties = properties;
        this.idGenerator = idGenerator;
    }

    @Transactional
    public UploadResult uploadLicense(LoginUser user, MultipartFile file) {
        if (file == null || file.isEmpty()) {
            throw new BizException(40001, "请选择营业执照文件");
        }
        if (file.getSize() > properties.getMaxSize()) {
            throw new BizException(40001, "营业执照文件不能超过 10MB");
        }
        String original = file.getOriginalFilename() == null ? "license" : file.getOriginalFilename();
        String ext = extension(original);
        if (ext == null || !ext.matches("(?i)png|jpe?g|pdf")) {
            throw new BizException(40001, "仅支持 JPG / PNG / PDF 格式的营业执照");
        }

        long fileId = idGenerator.nextId();
        String tenant = user.tenantCode() == null || user.tenantCode().isBlank()
                ? "PLATFORM"
                : user.tenantCode();
        String safeName = original.replaceAll("[\\\\/:*?\"<>|\\s]+", "_");
        String objectKey = LICENSE_PREFIX + "/" + tenant + "/" + fileId + "_" + safeName;

        try (InputStream in = file.getInputStream()) {
            objectStorage.upload(objectKey, in, file.getSize(), file.getContentType());
        } catch (IOException e) {
            throw new BizException(50001, "营业执照上传失败，请稍后重试");
        }

        String fileNo = "F" + fileId;
        fileMetaMapper.insert(FileMeta.builder()
                .id(fileId)
                .tenantCode(tenant)
                .fileNo(fileNo)
                .fileName(original)
                .objectKey(objectKey)
                .fileSize(file.getSize())
                .mimeType(file.getContentType())
                .bizType(2)
                .creator(String.valueOf(user.userId()))
                .deleted(false)
                .build());
        return new UploadResult(String.valueOf(fileId), original, file.getSize(), file.getContentType());
    }

    @Transactional(readOnly = true)
    public FileContent loadLicense(LoginUser user, long fileId) {
        FileMeta meta = fileMetaMapper.selectOne(
                Wrappers.<FileMeta>lambdaQuery()
                        .eq(FileMeta::getId, fileId)
                        .eq(FileMeta::getBizType, 2)
                        .eq(FileMeta::getDeleted, false)
                        .last("LIMIT 1")
        );
        if (meta == null) {
            throw new BizException(40401, "营业执照文件不存在");
        }
        return new FileContent(meta.getFileName(), meta.getMimeType(), objectStorage.download(meta.getObjectKey()));
    }

    private String extension(String name) {
        int dot = name.lastIndexOf('.');
        return dot < 0 || dot == name.length() - 1 ? null : name.substring(dot + 1).toLowerCase(Locale.ROOT);
    }

    public record UploadResult(String fileId, String fileName, long fileSize, String mimeType) {
    }

    public record FileContent(String fileName, String mimeType, InputStream inputStream) {
    }
}
```

```
async function loadGuide() {
  loading.value = true

    } else {

    }
  } catch (e) {
    loadError.value = e instanceof Error ? e.message : '加载失败，请稍后重试'
  } finally {
    loading.value = false
  }
}

function fillForm(state: EnterpriseGuideState) {

async function loadLicensePreview(fileId: string) {
  try {
    const blob = await fetchEnterpriseLicense(fileId)
    isImageLicense.value = blob.type.startsWith('image/')
    if (isImageLicense.value) {
      licensePreview.value = URL.createObjectURL(blob)
    }
  } catch {
    // 附件不可见时仍允许重新上传
  }
}

/**
 * 打开营业执照预览：图片/PDF 都会以新窗口打开，支持放大查看。
 */
async function openLicensePreview() {
  const fileId = guide.value?.licenseFileId
  if (!fileId) {
    ElMessage.warning('暂无可查看的营业执照')
    return
  }
  try {
    const blob = await fetchEnterpriseLicense(fileId)
    const url = URL.createObjectURL(blob)
    const link = document.createElement('a')
    link.href = url
    link.target = '_blank'
    link.rel = 'noopener'
    document.body.appendChild(link)
    link.click()
    link.remove()
  } catch {
    ElMessage.error('营业执照加载失败，请稍后重试')
  }
}

function pickLicense() {
  licenseInputRef.value?.click()
}

async function onLicenseChange(event: Event) {
  const input = event.target as HTMLInputElement
  const file = input.files?.[0]
  if (!file) return
  const typeOk = /\.(jpe?g|png|pdf)$/i.test(file.name)
  if (!typeOk) {
    ElMessage.warning('仅支持 JPG / PNG / PDF 格式的营业执照')
    return
  }
  if (file.size > 10 * 1024 * 1024) {
    ElMessage.warning('营业执照文件不能超过 10MB')
    return
  }
  try {
    const res = await uploadEnterpriseLicense(file)
    licenseFileId.value = res.fileId
    licenseFileName.value = res.fileName
    isImageLicense.value = /\.(jpe?g|png)$/i.test(res.fileName)
    if (isImageLicense.value) {
      licensePreview.value = URL.createObjectURL(file)
    }
    ElMessage.success('营业执照上传成功')
  } finally {
    input.value = ''
  }
}

function resetLicense() {
  licenseFileId.value = null
  licenseFileName.value = ''
  licensePreview.value = ''
  isImageLicense.value = false
}

async function handleSubmit() {
  if (!licenseFileId.value) {
    ElMessage.warning('请先上传营业执照扫描件')
    return
  }
  const valid = await formRef.value?.validate().catch(() => false)
  if (!valid) return
  submitting.value = true
  try {
    guide.value = await submitEnterpriseReview({
      companyName: form.companyName,
      industry: form.industry,
      scale: form.scale,
      licenseNo: form.licenseNo,
      registerAddress: form.registerAddress,
      legalPerson: form.legalPerson,
      licenseFileId: licenseFileId.value,
      contactName: form.contactName,
      contactPhone: form.contactPhone,
      contactEmail: form.contactEmail,
    })
    ElMessage.success('企业资料已提交，进入运营审核')
    startReviewTimer()
  } catch {
    // 错误提示由统一拦截器处理
  } finally {
    submitting.value = false
  }
}

function enterWorkspace() {
  localStorage.setItem(ONBOARD_DONE_KEY, 'true')
  router.push('/dashboard')
}

function doLogout() {
  userStore.reset()
  localStorage.removeItem(ONBOARD_DONE_KEY)
  router.push('/login')
}

async function openLogoutConfirm() {
  // 确保账号信息已恢复，弹窗里展示真实姓名与账号
  if (!userStore.userId) {
    try {
      await userStore.fetchProfile()
    } catch {
      // 拉取失败时按原信息弹窗，不阻断退出操作
    }
  }
  logoutVisible.value = true
}

function backToLogin() {
  userStore.reset()
  localStorage.removeItem(ONBOARD_DONE_KEY)
  router.push('/login')
}

function maskPhone(phone?: string) {
  if (!phone || phone.length < 11) return phone || '—'
  return \`${phone.slice(0, 3)}****${phone.slice(7)}\`
}

function startReviewTimer() {
  stopReviewTimer()
  reviewTimer = window.setInterval(async () => {
    try {
      const latest = await getEnterpriseGuide()
      if (latest.stage !== 'PENDING_REVIEW') {
        guide.value = latest
        stopReviewTimer()
      }
    } catch {
      // 轮询失败静默，下次重试
    }
  }, 30000)
}

function stopReviewTimer() {
  if (reviewTimer) {
    window.clearInterval(reviewTimer)
    reviewTimer = undefined
  }
}
```

## 八、常见问题

1. **上传接口返回 404** ：检查 gateway 是否已增加 /api/customer 路由，并检查 vite.config.ts 是否代理 /api/customer。
2. **下载文件 ID 不匹配** ：雪花 ID 超过 JS 安全整数，企业侧 licenseFileId 使用字符串传输；前端请勿再转成 number。
3. **MyBatis-Plus Mapper 注入失败** ：确认 customer-service 已创建 MybatisPlusAutoConfigBridge 与 mapper 目录。
